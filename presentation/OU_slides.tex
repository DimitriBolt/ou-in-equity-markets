\begin{frame}{Research Question and Financial Setup}
% ================================================================
\begin{block}{Central question}
Can stock-market time series be described by simple textbook SDEs?
\end{block}

\smallskip
\textbf{Financial setup} \textit{(for the SDE-oriented reader):}
\begin{itemize}
  \item A \textbf{stock} is a tradeable financial asset (share of a listed company);
        its \textbf{closing price} $P_t^k > 0$ is observed once per \textbf{trading day}
        ($\approx 252$ days/year).
  \item Dataset: 94 large-cap US stocks, NASDAQ, Jan 2023 -- Apr 2026.
\end{itemize}

\medskip
\textbf{Key insight:} the same price series yields five structurally different objects:
\[
P_{t}^{k}
\;\to\; X_{t}^{k} = \ln P_{t}^{k}
\;\to\; \Delta X_{t_i}^{k}
\;\to\; \widetilde{X}_{t}^{k}
\;\to\; \Delta\widetilde{X}_{t_i}^{k}
\]
A benchmark SDE may be useful for one transformation and inadequate for another.

\end{frame}

% ================================================================
\begin{frame}{Three Textbook Benchmarks ({\O}ksendal)}
% ================================================================

\small
Following \textbf{{\O}ksendal, \textit{Stochastic Differential Equations}, 6th ed.}

\medskip
\begin{enumerate}
  \setlength{\itemsep}{5pt}

  \item \textbf{It\^{o} diffusion} \hfill \textit{Def.\ 4.1.1, eq.\ (4.1.6)}
        \[ dX_t = u\,dt + v\,dB_t \]
        \vspace{-3pt}
        Benchmark for daily log-returns $\Delta X_{t_i}^k$.

  \item \textbf{Geometric Brownian Motion} \hfill \textit{Ch.\ 5.1, eq.\ (5.1.5)}
        \[ N_t = N_0\exp\!\bigl((r-\tfrac{1}{2}\alpha^2)t + \alpha B_t\bigr) \]
        \vspace{-3pt}
        $N_t$ = price level. Benchmark for raw log-prices $X_t^k$.

  \item \textbf{Ornstein--Uhlenbeck} \quad\textit{(Exercise 5.7)}
        \[ dX_t = \theta(m - X_t)\,dt + \sigma\,dB_t \]
        \vspace{-3pt}
        $\theta>0$: mean-reversion speed, $m$: equilibrium.
        Benchmark for detrended log-prices $\widetilde{X}_t^k$.
\end{enumerate}

\end{frame}

\begin{frame}{Data and Pipeline}

\begin{columns}[T]
\begin{column}{0.45\textwidth}
\textbf{Dataset}
\begin{itemize}\setlength{\itemsep}{1pt}
  \item 94 large-cap US stocks, daily prices

\smallskip
\textbf{Detrending:} OLS linear fit to $X_t^k$, keep residual:
\vspace{-4pt}\[ \widetilde{X}_{t}^k = X_{t}^k - (\hat a + \hat b\, t) \]\vspace{-4pt}
\small Different transformations $\to$ different benchmarks.
\end{column}
\begin{column}{0.50\textwidth}
\textbf{Five-stage pipeline}

\smallskip
\renewcommand{\arraystretch}{1.25}
\begin{tabular}{@{}ll@{}}
$P_{t}^{k}$ & prices \\
$\downarrow \ln$ & \\
$X_{t}^{k}$ & log-prices \;\small[\textit{GBM}] \\
$\downarrow \Delta$ & \\
$\Delta X_{t_i}^{k}$ & log-returns \;\small[\textit{It\^{o}}] \\
$\downarrow$ OLS detrend & \\
$\widetilde{X}_{t}^{k}$ & detrended levels \;\small[\textit{OU}] \\
$\downarrow \Delta$ & \\
$\Delta\widetilde{X}_{t_i}^{k}$ & detrended returns \\
\end{tabular}
\end{column}

\begin{frame}{OU Parameter Estimation: AR(1) Proxy}
% ================================================================

\textbf{Bridge from continuous OU to discrete daily data:}
\[
dX_t = \theta(m - X_t)\,dt + \sigma\,dB_t
\;\xrightarrow{\;\text{one step}\;}\;
\widetilde{X}_{t+1} \approx \hat\alpha + \hat\phi\,\widetilde{X}_{t} + \varepsilon_t
\]
\small $\hat\phi = e^{-\hat\theta}$,\quad $\hat m = \hat\alpha/(1-\hat\phi)$,\quad $t_{1/2} = \ln 2/\hat\theta$

\medskip
\begin{center}
\small\renewcommand{\arraystretch}{1.2}
\begin{tabular}{lccccc}
\toprule
Symbol & $\hat\phi$ & $\hat\theta$ & $t_{1/2}$ (days) & $\hat m$ & $\hat\sigma$ \\
\midrule
\rowcolor{lightgray}
NVDA & 0.9878 & 0.0123 & \textbf{56.4} & 0.0337 & 0.0305 \\
META & 0.9879 & 0.0122 & \textbf{56.8} & 0.0245 & 0.0236 \\
\rowcolor{lightgray}
GOOG & 0.9880 & 0.0121 & \textbf{57.4} & 0.0207 & 0.0187 \\
\bottomrule
\end{tabular}
\end{center}

\medskip\small
\begin{itemize}\setlength{\itemsep}{1pt}
  \item $\hat\phi < 1$: valid slow mean reversion (not unit root)
  \item $t_{1/2} \approx 57$ trading days $\approx$ \textbf{3 calendar months}: gentle, gradual drift
\end{itemize}

\end{frame}

\begin{frame}{Model Assessment Criterion}
% ================================================================
\begin{alertblock}{What does ``compatible'' mean?}
An explicit criterion is needed --- visual similarity alone is not sufficient.
\end{alertblock}

\smallskip
\textbf{Assessment workflow:}
\begin{enumerate}\setlength{\itemsep}{2pt}

  \item \textbf{Fit} using one simple statistic:
        \begin{itemize}
          \item One-step AR(1) on detrended levels $\;\Rightarrow\;$ $\hat\alpha,\;\hat\phi$
        \end{itemize}

  \item \textbf{Validate} against implications \textbf{not used} in the fit:
        \begin{itemize}
          \item ACF decay: empirical ACF at lags $2$--$20$ vs.\ predicted $\hat\phi^{\ell}$
          \item Transition mean: $\mathbb{E}[\widetilde{X}_{t+h}\mid\widetilde{X}_t]$ at $h=5$
                vs.\ model prediction $\hat m + \hat\phi^h(\widetilde{X}_t - \hat m)$
        \end{itemize}

  \item \textbf{Report the known failure:}
        time-varying volatility (rolling-vol ratio $> 3\times$) lies outside the model.
\end{enumerate}

\end{frame}

% ================================================================

\begin{center}
\small\renewcommand{\arraystretch}{1.1}
\begin{tabular}{l|cc|c|cc|c}
\toprule
 & \multicolumn{2}{c|}{\textit{Fit}} & &
   \multicolumn{2}{c|}{\textit{Validation}} & \textit{Failure} \\
Sym. & Raw $R^2$ & OU $R^2$ & $t_{1/2}$ & ACF RMSE & Trans.\ RMSE & Vol.\ ratio \\
\midrule
\rowcolor{lightgray}
NVDA & 0.883 & 0.983 & 56.4\,d & 0.049 & 0.0055 & $4.1\times$ \\
META & 0.822 & 0.984 & 56.8\,d & 0.038 & 0.0052 & $6.8\times$ \\
\rowcolor{lightgray}
GOOG & 0.866 & 0.976 & 57.4\,d & \textbf{0.007} & 0.0069 & $3.7\times$ \\
\bottomrule
\end{tabular}
\end{center}

\smallskip\small
\begin{itemize}\setlength{\itemsep}{0pt}
  \item \textbf{Raw $R^2$ high} --- GBM benchmark applies to log-prices.
  \item \textbf{OU $R^2 \approx 0.98$} --- one-step AR(1) fit is very tight.
  \item \textbf{ACF RMSE small} (GOOG: 0.007) --- $\hat\phi^\ell$ tracks empirical decay. \textbf{Validation 1 passes.}
  \item \textbf{Trans.\ RMSE $\approx 0.005$--$0.007$} --- conditional mean accurate. \textbf{Validation 2 passes.}
  \item \textbf{Vol.\ ratio $> 3\times$} --- constant-$\sigma$ SDE fails here (expected).
\end{itemize}

\end{frame}

\begin{frame}{Main Findings}
% ================================================================

\textbf{The answer is qualified, not absolute:}

\medskip
\begin{enumerate}\setlength{\itemsep}{8pt}

  \item \textbf{GBM $\Rightarrow$ raw log-prices} ($R^2\approx 0.82$--$0.88$)\\
        \small Raw log-prices are trend-dominated; multiplicative-growth benchmark applies.

  \item \textbf{OU $\Rightarrow$ detrended log-prices} (lag-1 ACF $\approx 0.988$--$0.992$)\\
        \small Half-life $t_{1/2}\approx 57$ days ($\approx 3$ months).
        ACF-decay and conditional-mean validations both pass.

  \item \textbf{It\^{o} diffusion $\Rightarrow$ increments, but $\sigma$ is not constant}\\
        \small Rolling-vol ratio $>3$--$7\times$: volatility clustering is outside the model.

\end{enumerate}

\medskip\small
No single low-dimensional SDE simultaneously explains all features.

\end{frame}

% ================================================================
\begin{frame}{Limitations and Natural Extensions}
% ================================================================

\begin{columns}[T]
\begin{column}{0.47\textwidth}
\textbf{Current limitations}
\begin{enumerate}\setlength{\itemsep}{3pt}\small
  \item Educational diagnostic, not formal likelihood comparison
  \item Simple linear detrending (not adaptive, not factor-based)
  \item Daily data only
  \item Constant-$\sigma$ misses \emph{volatility clustering}
  \item Transition-mean check is simplified; not a full Markov transition-kernel test
\end{enumerate}
\end{column}
\begin{column}{0.50\textwidth}
\textbf{Natural next steps}
\begin{enumerate}\setlength{\itemsep}{3pt}\small
  \item Realized quadratic variation: directly measures $\langle X\rangle_t$
  \item PCA market-factor extraction; OU on detrended factor
  \item Stochastic-volatility extension ($\sigma_t$ follows a diffusion)
  \item Jump-diffusion models for large overnight moves
  \item Fokker--Planck and full transition-kernel analysis
  \item MLE-based parameter estimation and formal model comparison
\end{enumerate}
\end{column}
\end{columns}

\end{frame}

% ================================================================
\begin{frame}{Conclusion}
% ================================================================
\begin{block}{Answer}
Stock-market time series \textbf{can} be described by SDEs to a meaningful extent,
but the description is \emph{approximate} and depends on which object is modeled.
\end{block}

\smallskip
\begin{itemize}\setlength{\itemsep}{2pt}
  \item Raw $X_t^k$ (log-prices) $\;\to\;$ \textbf{GBM} is the appropriate benchmark.
  \item Detrended $\widetilde{X}_t^k$ $\;\to\;$ \textbf{OU} with $t_{1/2}\approx 57\,\text{d}$.
  \item Increments $\Delta X_{t_i}^k$ $\;\to\;$ \textbf{It\^{o} diffusion} language is natural,
        but $\sigma$ is time-varying.
\end{itemize}

\smallskip
\begin{alertblock}{Methodological lesson}
``Compatible'' = fit by one set of statistics, validate by others.
\end{alertblock}
